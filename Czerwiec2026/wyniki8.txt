SELECT kategorie.Nazwa_Kategorii, SUM(oceny.Ocena)
FROM kategorie
INNER JOIN oceny
ON kategorie.Id_Kategorii = oceny.Id_Kategorii
GROUP BY kategorie.Nazwa_Kategorii;

8.5
SELECT TOP 1 uczniowie.Id_Ucznia, Count(oceny.Ocena)
FROM uczniowie
INNER JOIN kategorie
INNER JOIN oceny ON kategorie.Id_Kategorii = oceny.Id_Kategorii
ON uczniowie.Id_Ucznia = oceny.Id_Ucznia
WHERE kategorie.Nazwa_Kategorii = "sprawdzian" AND oceny.Ocena = 6
GROUP BY uczniowie.Id_Ucznia
ORDER BY Count(oceny.Ocena) DESC;
